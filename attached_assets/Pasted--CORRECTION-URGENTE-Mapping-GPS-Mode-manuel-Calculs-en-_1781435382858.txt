1. DISTANCE DEPUIS LE POINT PRÉCÉDENT
   Utiliser la formule HAVERSINE :

   function haversine(lat1, lng1,
                      lat2, lng2) {
     const R = 6371000 // rayon Terre en m
     const φ1 = lat1 * Math.PI/180
     const φ2 = lat2 * Math.PI/180
     const Δφ = (lat2-lat1) * Math.PI/180
     const Δλ = (lng2-lng1) * Math.PI/180
     const a = Math.sin(Δφ/2)²
               + Math.cos(φ1)
               * Math.cos(φ2)
               * Math.sin(Δλ/2)²
     return R * 2 * Math.atan2(
       Math.sqrt(a),
       Math.sqrt(1-a)
     )
   }

   function superficie(points) {
     // Convertir en mètres
     // depuis le premier point
     const ref = points[0]
     const R = 6371000
     const pts = points.map(p => ({
       x: haversine(ref.lat, ref.lng,
                    ref.lat, p.lng),
       y: haversine(ref.lat, ref.lng,
                    p.lat, ref.lng)
     }))

     // Shoelace
     let area = 0
     const n = pts.length
     for(let i = 0; i < n; i++) {
       const j = (i + 1) % n
       area += pts[i].x * pts[j].y
       area -= pts[j].x * pts[i].y
     }
     return Math.abs(area / 2)
       / 10000 // en hectares
   }
